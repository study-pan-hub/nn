# ============================================================
# 安装说明
# ============================================================
#
# 1. 基础安装：
#    pip install -r requirements.txt
#
# 2. PyTorch GPU 版本安装（推荐）：
#    pip install torch --index-url https://download.pytorch.org/whl/cu118
#
# 3. CARLA 安装：
#    - 下载 CARLA 仿真器：https://github.com/carla-simulator/carla/releases
#    - 推荐版本：CARLA 0.9.14
#    - 将 CARLA Python API 添加到 Python 路径：
#      export PYTHONPATH=$PYTHONPATH:/path/to/CARLA/PythonAPI/carla/dist/carla-*.egg
#
# 4. ROS 1 安装（Ubuntu 20.04 Noetic）：
#    sudo apt install ros-noetic-desktop-full
#    source /opt/ros/noetic/setup.bash
#
# 5. 构建 ROS 包：
#    cd ~/catkin_ws
#    catkin_make
#    source devel/setup.bash
#
# 6. 运行项目：
#    - 训练：         python train_agent.py --timesteps 300000
#    - 评估：         python eval_agent.py --model_path ./checkpoints/best_model.zip
#    - ROS 训练节点： rosrun carla_rl_ros ros_train_node.py
#    - ROS 评估节点： rosrun carla_rl_ros ros_eval_node.py
#    - Launch 文件：  roslaunch carla_rl_ros eval.launch steps:=1000
#
